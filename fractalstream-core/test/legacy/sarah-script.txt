dyn [Dynamics]
dyn iterate z^2 + c until z escapes.
dyn report z.
par block
set b to a.
set a to z.
set z to z^2 + c.
until z escapes or z vanishes.
report z.
if z vanishes then
if (a-c) < (-a-c) then
[BA]
if (b-c) < (-b-c) then [AA] end.
end.
if (-a-c) < (a-c) then
[BB]
if (b-c) < (-b-c) then [AB] end.
end.
end.
end.
